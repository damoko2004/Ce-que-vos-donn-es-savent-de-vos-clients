-- Volumes par canal, hier contre la moyenne des quatre semaines.
SELECT canal,
       count(*) FILTER (WHERE date_commande = current_date - 1)  AS hier,
       count(*) FILTER (WHERE date_commande BETWEEN current_date - 29
                                              AND current_date - 2) / 28.0 AS moyenne
FROM raw_commandes
GROUP BY canal
ORDER BY canal;
